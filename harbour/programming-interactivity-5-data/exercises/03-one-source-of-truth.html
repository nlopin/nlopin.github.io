Act 3 · State

Single source of truth

A filter, a search box and a ♡ on every card: the three features that conflicted in the jam. This time they all update one state object, and one render() renders the DOM from it. drawCards(list, saved) is given (🔒).

The loop. An event changes state, then calls render(). render() reads only state and renders everything: cards, pressed buttons, counts. Nothing reads state back from the DOM.

1

The filter, from state

In the jam, the filter hid cards with hidden, and every other feature fought it.

This time nothing is hidden: the list is rendered from state. A click only sets state.tag and calls render(). visibleStalls() decides what's on screen; render() also marks the pressed button.

Solution

visibleStalls() derives the list from state: nothing is stored twice. render() renders the cards and the buttons, so the pressed button can never disagree with state.tag.

const state = {
  tag: "all",
};

const filters = document.querySelector(".filters");

function visibleStalls() {
  return stalls.filter((s) => state.tag === "all" || s.tag === state.tag);
}

function render() {
  drawCards(visibleStalls(), new Set());
  for (const b of filters.querySelectorAll("button")) {
    b.setAttribute("aria-pressed", String(b.dataset.filter === state.tag));
  }
}

filters.addEventListener("click", (event) => {
  const button = event.target.closest("button");
  if (!button) return;
  state.tag = button.dataset.filter;
  render();
});

render();
2

Search, and the count

“Is there a taco stall?” The filter alone can't answer that.

Add query to the state. Typing in #q sets it on every keystroke: the input event (the checks fire input, so keyup won't pass). visibleStalls() applies the tag and the query (a name that contains it, upper or lower case). render() writes "3 of 8 stalls" into .count.

Solution

Two independent controls, and neither knows about the other: both only write to state. Filtering twice in a row is fine; includes("") is always true, so an empty box matches everything.

const state = {
  tag: "all",
  query: "",
};

const filters = document.querySelector(".filters");

function visibleStalls() {
  const q = state.query.trim().toLowerCase();
  return stalls
    .filter((s) => state.tag === "all" || s.tag === state.tag)
    .filter((s) => s.name.toLowerCase().includes(q));
}

function render() {
  const visible = visibleStalls();
  drawCards(visible, new Set());
  for (const b of filters.querySelectorAll("button")) {
    b.setAttribute("aria-pressed", String(b.dataset.filter === state.tag));
  }
  document.querySelector(".count").textContent = `${visible.length} of ${stalls.length} stalls`;
}

filters.addEventListener("click", (event) => {
  const button = event.target.closest("button");
  if (!button) return;
  state.tag = button.dataset.filter;
  render();
});

document.querySelector("#q").addEventListener("input", (event) => {
  state.query = event.target.value;
  render();
});

render();
3

Save a stall ♥

The saved flag is lost as soon as a filter re-renders the list.

Add saved: new Set() to the state. One listener on .vendor-list (the cards are redrawn all the time: listeners on them would die with them): a click on a .save adds or removes that stall's id. Pass state.saved to drawCards, and write "♥ 2" into .hearts.

Solution

The cards are thrown away on every render, so a listener on a card dies with it. One listener on the list (delegation, Lecture 4) hears every ♡, now and after any redraw. The ♥ isn't stored in the button: the button is rendered from state.saved.

const state = {
  tag: "all",
  query: "",
  saved: new Set(),
};

const filters = document.querySelector(".filters");

function visibleStalls() {
  const q = state.query.trim().toLowerCase();
  return stalls
    .filter((s) => state.tag === "all" || s.tag === state.tag)
    .filter((s) => s.name.toLowerCase().includes(q));
}

function render() {
  const visible = visibleStalls();
  drawCards(visible, state.saved);
  for (const b of filters.querySelectorAll("button")) {
    b.setAttribute("aria-pressed", String(b.dataset.filter === state.tag));
  }
  document.querySelector(".count").textContent = `${visible.length} of ${stalls.length} stalls`;
  document.querySelector(".hearts").textContent = `♥ ${state.saved.size}`;
}

filters.addEventListener("click", (event) => {
  const button = event.target.closest("button");
  if (!button) return;
  state.tag = button.dataset.filter;
  render();
});

document.querySelector("#q").addEventListener("input", (event) => {
  state.query = event.target.value;
  render();
});

list.addEventListener("click", (event) => {
  const button = event.target.closest(".save");
  if (!button) return;
  const id = button.closest(".stall").dataset.id;
  if (state.saved.has(id)) state.saved.delete(id);
  else state.saved.add(id);
  render();
});

render();

Only my saved stalls

A checkbox, #only-saved. Ticked: only saved stalls, still combined with the filter and the search. One more key in the state, one more filter.

Solution

The change event on a checkbox, event.target.checked into the state, render. visibleStalls() gets one more line. Nothing else moves.

const state = {
  tag: "all",
  query: "",
  saved: new Set(),
  onlySaved: false,
};

const filters = document.querySelector(".filters");

function visibleStalls() {
  const q = state.query.trim().toLowerCase();
  return stalls
    .filter((s) => state.tag === "all" || s.tag === state.tag)
    .filter((s) => s.name.toLowerCase().includes(q))
    .filter((s) => !state.onlySaved || state.saved.has(s.id));
}

function render() {
  const visible = visibleStalls();
  drawCards(visible, state.saved);
  for (const b of filters.querySelectorAll("button")) {
    b.setAttribute("aria-pressed", String(b.dataset.filter === state.tag));
  }
  document.querySelector(".count").textContent = `${visible.length} of ${stalls.length} stalls`;
  document.querySelector(".hearts").textContent = `♥ ${state.saved.size}`;
}

filters.addEventListener("click", (event) => {
  const button = event.target.closest("button");
  if (!button) return;
  state.tag = button.dataset.filter;
  render();
});

document.querySelector("#q").addEventListener("input", (event) => {
  state.query = event.target.value;
  render();
});

document.querySelector("#only-saved").addEventListener("change", (event) => {
  state.onlySaved = event.target.checked;
  render();
});

list.addEventListener("click", (event) => {
  const button = event.target.closest(".save");
  if (!button) return;
  const id = button.closest(".stall").dataset.id;
  if (state.saved.has(id)) state.saved.delete(id);
  else state.saved.add(id);
  render();
});

render();
