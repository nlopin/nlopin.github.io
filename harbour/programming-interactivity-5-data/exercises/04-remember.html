Remember my night

I saved four stalls on my phone, reloaded the page at the market, and they were gone.

Everything works, until a reload. Write save() and load(): the state worth persisting (the filter and the saved ids) go to localStorage under the key "my-night", as JSON. Decide where to call them. Then test invalid data: DevTools → Application → Local storage, change the stored value into invalid JSON, press ▶ run. (Reloading the whole page is fine too: your code is kept as a draft until you press ↺ reset.) (A broken earlier attempt left junk in storage? )

Solution

localStorage keeps strings only, so the state goes through JSON.stringify, and a Set has to become an array first (JSON.stringify(new Set(["a"])) is "{}"). Save after every change: one update() that renders and saves, called from every listener. Load once, before the first render. Everything that comes back from storage is untrusted: missing, old, or broken. try … catch and ?? keep the defaults.

const state = {
  tag: "all",
  query: "",
  saved: new Set(),
};

const filters = document.querySelector(".filters");

function visibleStalls() {
  const q = state.query.trim().toLowerCase();
  return stalls
    .filter((s) => state.tag === "all" || s.tag === state.tag)
    .filter((s) => s.name.toLowerCase().includes(q));
}

function render() {
  const visible = visibleStalls();
  drawCards(visible, state.saved);
  for (const b of filters.querySelectorAll("button")) {
    b.setAttribute("aria-pressed", String(b.dataset.filter === state.tag));
  }
  document.querySelector(".count").textContent = `${visible.length} of ${stalls.length} stalls`;
  document.querySelector(".hearts").textContent = `♥ ${state.saved.size}`;
}

filters.addEventListener("click", (event) => {
  const button = event.target.closest("button");
  if (!button) return;
  state.tag = button.dataset.filter;
  update();
});

document.querySelector("#q").addEventListener("input", (event) => {
  state.query = event.target.value;
  update();
});

list.addEventListener("click", (event) => {
  const button = event.target.closest(".save");
  if (!button) return;
  const id = button.closest(".stall").dataset.id;
  if (state.saved.has(id)) state.saved.delete(id);
  else state.saved.add(id);
  update();
});

// ---- remember ------------------------------------------------
const KEY = "my-night";

function save() {
  const data = { tag: state.tag, saved: [...state.saved] };
  localStorage.setItem(KEY, JSON.stringify(data));
}

function load() {
  try {
    const data = JSON.parse(localStorage.getItem(KEY));
    if (!data) return; // nothing saved yet
    state.tag = data.tag ?? "all";
    state.saved = new Set(data.saved ?? []);
  } catch {
    // broken JSON: keep the defaults
  }
}

// every change: draw, then remember
function update() {
  render();
  save();
}

load();
render();

Clear my night

A #clear button: back to the defaults (all stalls, nothing saved, an empty search box), on screen and in storage. Notice how little code it takes when everything goes through state.

Solution

Set every key back to its default and call update(). The search box is a control the state also describes, so render() writes state.query back into it (only when it differs: rewriting an input you're typing in can move the cursor).

const state = {
  tag: "all",
  query: "",
  saved: new Set(),
};

const filters = document.querySelector(".filters");

function visibleStalls() {
  const q = state.query.trim().toLowerCase();
  return stalls
    .filter((s) => state.tag === "all" || s.tag === state.tag)
    .filter((s) => s.name.toLowerCase().includes(q));
}

function render() {
  const visible = visibleStalls();
  drawCards(visible, state.saved);
  for (const b of filters.querySelectorAll("button")) {
